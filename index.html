//+------------------------------------------------------------------+ //| RoyaltyFX_ALL.mq5 | //| ROYALTY FX PRO - REAL TIME ALL SYMBOLS AUTO TRADER | //+------------------------------------------------------------------+ #property copyright "Royalty FX Pro" #property version "2.00" input bool AutoTradeAll = true; input double Lots = 0.05; input int MagicNumber = 20251102; input int StopLossPoints = 800; input int TakeProfitPoints = 1500; input int MaxTradesPerSymbol= 1; input int RSI_Period = 14; input int RSI_Overbought = 70; input int RSI_Oversold = 30; input bool UseTrendFilter = true; input int EMA_Fast = 21; input int EMA_Slow = 50; string Symbols[18] = { "XAUUSD","XAGUSD","EURUSD","GBPUSD","USDJPY","USDCHF", "AUDUSD","NZDUSD","USDCAD","BTCUSD","ETHUSD","BNBUSD", "US30","NAS100","SPX500","GER40","UK100","USOIL" }; int rsi_handles[18]; int ema_fast_handles[18]; int ema_slow_handles[18]; int OnInit() { for(int i=0;i<18;i++) { SymbolSelect(Symbols[i], true); rsi_handles[i] = iRSI(Symbols[i], PERIOD_M15, RSI_Period, PRICE_CLOSE); ema_fast_handles[i] = iMA(Symbols[i], PERIOD_M15, EMA_Fast, 0, MODE_EMA, PRICE_CLOSE); ema_slow_handles[i] = iMA(Symbols[i], PERIOD_M15, EMA_Slow, 0, MODE_EMA, PRICE_CLOSE); } EventSetTimer(10); return(INIT_SUCCEEDED); } void OnDeinit(const int reason) { EventKillTimer(); for(int i=0;i<18;i++) { IndicatorRelease(rsi_handles[i]); IndicatorRelease(ema_fast_handles[i]); IndicatorRelease(ema_slow_handles[i]); } } void OnTimer() { if(!AutoTradeAll) return; ScanAllSymbols(); } void ScanAllSymbols() { for(int i=0;i<18;i++) { string sym = Symbols[i]; if(!IsNewBar(sym, PERIOD_M15)) continue; if(CountTrades(sym) >= MaxTradesPerSymbol) continue; double rsi_val[1], ema_fast[1], ema_slow[1]; if(CopyBuffer(rsi_handles[i],0,0,1,rsi_val)<=0) continue; if(CopyBuffer(ema_fast_handles[i],0,0,1,ema_fast)<=0) continue; if(CopyBuffer(ema_slow_handles[i],0,0,1,ema_slow)<=0) continue; bool upTrend = ema_fast[0] > ema_slow[0]; bool downTrend = ema_fast[0] < ema_slow[0]; if(rsi_val[0] < RSI_Oversold && (!UseTrendFilter || upTrend)) OpenTrade(sym, ORDER_TYPE_BUY); else if(rsi_val[0] > RSI_Overbought && (!UseTrendFilter || downTrend)) OpenTrade(sym, ORDER_TYPE_SELL); } } void OpenTrade(string sym, ENUM_ORDER_TYPE type) { MqlTick tick; if(!SymbolInfoTick(sym, tick)) return; double price = (type==ORDER_TYPE_BUY) ? tick.ask : tick.bid; double point = SymbolInfoDouble(sym, SYMBOL_POINT); double sl = (type==ORDER_TYPE_BUY) ? price - StopLossPoints*point : price + StopLossPoints*point; double tp = (type==ORDER_TYPE_BUY) ? price + TakeProfitPoints*point : price - TakeProfitPoints*point; MqlTradeRequest req; MqlTradeResult res; ZeroMemory(req); req.action = TRADE_ACTION_DEAL; req.symbol = sym; req.volume = Lots; req.type = type; req.price = price; req.sl = sl; req.tp = tp; req.deviation = 30; req.magic = MagicNumber; req.type_filling = ORDER_FILLING_IOC; req.comment = "RoyaltyFX Auto"; OrderSend(req,res); } int CountTrades(string sym) { int count=0; for(int i=PositionsTotal()-1;i>=0;i--) { ulong ticket = PositionGetTicket(i); if(PositionGetString(POSITION_SYMBOL)==sym && PositionGetInteger(POSITION_MAGIC)==MagicNumber) count++; } return count; } bool IsNewBar(string sym, ENUM_TIMEFRAMES tf) { static datetime lastBar[18]; int idx = -1; for(int i=0;i<18;i++) if(Symbols[i]==sym) idx=i; if(idx==-1) return true; datetime curBar = iTime(sym, tf, 0); if(curBar != lastBar[idx]) { lastBar[idx]=curBar; return true; } return false; }
